   const getCurrentAcademicYear = (years: any[]) => {
     let current = years.find(y => y.is_current === true);
     if (!current) {
       const today = new Date();
       current = years.find(y => {
         const startDate = new Date(y.start_date);
         const endDate = new Date(y.end_date);
         return today >= startDate && today <= endDate;
       });
     }
     return current;
   };
   
   // Determinar si es año siguiente
   const isNextYear = (year: any) => {
     const today = new Date();
     const startDate = new Date(year.start_date);
     const currentYear = getCurrentAcademicYear(academicYears);
     return year.id !== currentYear?.id && startDate > today;
   };
   
   // Filtrar años futuros
   const today = new Date();
   const futureAcademicYears = academicYears.filter(year => {
     const endDate = new Date(year.end_date);

   });

7. ✅ handleSubmit - OPCIÓN B (Ambas API calls)
   ```typescript
   const handleSubmit = async (e: React.FormEvent) => {
     e.preventDefault();
     
     // Validaciones...
     if (!formData.subjectId || !formData.sectionId || !formData.academicYearId) {
       toast.error("Por favor completa los campos básicos");
       return;
     }
     
     if (!formData.weeksDuration) {
       toast.error("Duración en semanas es requerida");
       return;
     }
     
     const weeks = parseInt(formData.weeksDuration);
     if (isNaN(weeks) || weeks < 1 || weeks > 52) {
       toast.error("Duración debe estar entre 1 y 52 semanas");
       return;
     }
     
     if (formData.schedules.length === 0) {
       toast.error("Por favor agrega al menos un horario");
       return;
     }

     setIsLoading(true);
     try {
       // PASO 1: Crear clase
       const classResponse = await apiClient.classes.createClass({
         subjectId: formData.subjectId,
         sectionId: formData.sectionId,
         academicYearId: formData.academicYearId,
         classroom: formData.classroom || "",
         weeksDuration: weeks
       });

       if (!classResponse.success || !classResponse.data?.id) {
         throw new Error("Error al crear la clase");
       }

       // PASO 2: Crear horarios
       const schedulesResponse = await apiClient.classes.createSchedules(
         classResponse.data.id,
         { schedules: formData.schedules }
       );

       if (!schedulesResponse.success) {
         throw new Error("Error al guardar los horarios");
       }

       // ✅ TODO OK
       onSave(classResponse.data);
       
       setFormData({
         subjectId: "",
         sectionId: "",
         academicYearId: "",
         classroom: "",
         weeksDuration: "",
         schedules: []
       });
       
       toast.success("Clase y horarios creados exitosamente");
       onClose();

     } catch (error) {
       console.error("Error:", error);
       toast.error(`Error: ${error instanceof Error ? error.message : "Desconocido"}`);
       // Modal NO cierra → usuario puede corregir
     } finally {
       setIsLoading(false);
     }
   };
   ```
